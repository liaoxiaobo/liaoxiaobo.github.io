# 直播间压力测试方案

# 一、压测目标

模拟1000用户登陆直播间，同时在线听课的场景，重点关注直播间的稳定性、音画质量。

# 二、压测环境准备

```Bash
for i in $(seq 4444 4453);do docker run -d -p $i:4444 --shm-size="4g" -e SE_NODE_MAX_SESSIONS=30 -e SE_NODE_SESSION_TIMEOUT=18000 selenium/standalone-chrome:4.10.0-20230607;done
```

以上命令是在物理机上批量创建多个 Selenium Chrome 节点容器，每个容器映射到不同的主机端口。下面是对每个参数的详细解释：

## **Docker 运行参数**

- `-d`
`--detach` 的简写，表示在后台运行容器（分离模式）

- `-p $i:4444`
`--publish` 的简写，将容器的 4444 端口映射到主机的 `$i` 端口（从 4444 到 4453）

    - 格式：`主机端口:容器端口`

    - Selenium Grid 节点默认使用 4444 端口

- `--shm-size="4g"`
设置容器的共享内存大小为 4GB

    - Chrome 浏览器需要较大的共享内存空间来正常运行

    - 默认的 64MB 通常不够，可能导致浏览器崩溃

## **环境变量参数**

- `-e SE_NODE_MAX_SESSIONS=30`
设置每个节点最大并发会话数为 30

    - 即每个容器可以同时处理最多 30 个浏览器实例

    - 实际值应根据服务器资源和需求调整

- `-e SE_NODE_SESSION_TIMEOUT=18000`
设置会话超时时间为 18000 秒（5 小时）

    - 如果一个会话空闲超过此时间，将被自动终止

    - 默认值为 300 秒（5 分钟）

## **镜像参数**

- `selenium/standalone-chrome:4.10.0-20230607`
使用的 Docker 镜像名称和标签

    - 这是 Selenium 官方提供的 Chrome 独立节点镜像

    - 版本 4\.10\.0\-20230607 表示 Selenium Grid 4\.10\.0 版本，构建于 2023 年 6 月 7 日

# 三、压测脚本

## 压测场景一：多用户登录直播间

**如下常量需要提前配置正确**

- `BASE_URL`：直播平台的登录页面URL。

- `URL_PARAMS`：URL参数，包括房间标题、房间ID、班级ID、用户角色和签名等。

- `SE_NODES`：Selenium Grid节点的URL列表，支持多个节点以实现分布式测试。

- `SE_NODE_SESSIONS`：每个节点上创建的浏览器会话数量。

```Python

from selenium.webdriver.common.keys import Keys
from selenium.webdriver.remote.webdriver import WebDriver
from selenium.common.exceptions import NoSuchElementException, InvalidSelectorException

import random
import string


class LiveStreamTester:
    *"""直播平台压力测试工具"""*

*    *# 常量配置
    BASE_URL = "https://live.soa.com.cn:55555/#/login"
    URL_PARAMS = {
        "room_title": "Stress_Test",
        "room_id": "7626-5831-7513",
        "classId": "1939963504730525698",
        "user_role": "10",
        "sign": "U2FsdGVkX1+pLUjaZO8mdqoitBaOqupZuwf9veBmMbkV8OxkOapzZH9oaZH4GjjJ"
    }
    SE_NODES = ["http://172.22.1.93:4444", "http://172.22.1.93:4445"]
    SE_NODE_SESSIONS = 30
    NAME_PREFIX = "autotest-"

    def __init__(self):
        *"""初始化测试器"""*
*        *self._setup_webdriver_options()

    def _setup_webdriver_options(self):
        *"""配置WebDriver选项"""*
*        *self.chrome_options = webdriver.ChromeOptions()
        self.chrome_options.add_experimental_option("detach", True)  # 不自动关闭浏览器
        self.chrome_options.add_argument("ignore-certificate-errors")  # 忽略SSL证书错误

    def _generate_username(self, length: int = 5) -> str:
        *"""生成随机用户名"""*
*        *chars = string.ascii_lowercase
        suffix = "".join(random.choice(chars) for _ in range(length))
        return f"{self.NAME_PREFIX}{suffix}"

    def _construct_url(self) -> str:
        *"""构建完整的测试URL"""*
*        *params = "&".join(f"{k}={v}" for k, v in self.URL_PARAMS.items())
        return f"{self.BASE_URL}?{params}"

    def _login(self, driver: WebDriver, username: str) -> bool:
        *"""执行登录流程"""*
*        *try:
            # 访问目标URL
            driver.get(self._construct_url())
            driver.implicitly_wait(10)
            time.sleep(1)  # 等待页面加载完成

            # 输入用户名
            username_input = self._find_element_safely(driver, "//div[4]/div/div/div/input")
            username_input.send_keys(username)

            # 点击登录按钮

            login_button.click()

            print(f"用户 {username} 登录成功!")
            return True

        except Exception as e:
            print(f"用户 {username} 登录失败: {str(e)}")
            return False

    def _chat(self, driver: WebDriver, username: str):
        *"""执行聊天交互"""*
*        *try:
            # 找到聊天输入框并输入消息
            chat_input = self._find_element_safely(driver, ".textarea-container:nth-child(2) > .textarea")
            chat_input.click()
            chat_input.send_keys(f"用户 {username} 发来一条弹幕！")
            chat_input.send_keys(Keys.ENTER)

            print(f"用户 {username} 发送弹幕成功!")

        except Exception as e:
            print(f"用户 {username} 发送弹幕失败: {str(e)}")

    def _find_element_safely(self, driver: WebDriver, selector: str):
        *"""*
*        安全查找元素（单个选择器版本）*

*        参数:*
*            driver: WebDriver实例*
*            selector: 元素选择器字符串（CSS或XPath）*

*        返回:*
*            WebElement: 找到的元素*

*        异常:*
*            NoSuchElementException: 元素未找到*
*            InvalidSelectorException: 选择器无效*
*        """*
*        *try:
            # 自动判断选择器类型
            if selector.startswith(("//", "./", "(")):
                return driver.find_element(By.XPATH, selector)
            return driver.find_element(By.CSS_SELECTOR, selector)
        except InvalidSelectorException as e:
            raise InvalidSelectorException(f"无效的选择器: '{selector}'. 错误详情: {str(e)}")
        except NoSuchElementException as e:
            raise NoSuchElementException(f"未找到元素: '{selector}'. 错误详情: {str(e)}")

    def run_stress_test(self):
        *"""执行压力测试"""*
*        *test_url = self._construct_url()
        print(f"开始压力测试，目标URL: {test_url}")

        for node_index, node_url in enumerate(self.SE_NODES):
            for instance_num in range(self.SE_NODE_SESSIONS):
                username = f"{self._generate_username()}{node_index * self.SE_NODE_SESSIONS + instance_num}"
                selenium_url = f"{node_url}/wd/hub"

                print(f"创建浏览器会话 (节点:{node_url}, 用户名:{username})")

                try:
                    driver = webdriver.Remote(
                        command_executor=selenium_url,
                        options=self.chrome_options
                    )
                    if self._login(driver, username):
                        self._chat(driver, username)
                except Exception as e:
                    print(f"创建WebDriver实例失败:{str(e)}")


if __name__ == "__main__":
    tester = LiveStreamTester()
    tester.run_stress_test()
```

## 压测场景二：多用户同时发送消息

**如下常量需要提前配置正确**

- `BASE_URL`：直播平台的登录页面URL。

- `URL_PARAMS`：URL参数，包括房间标题、房间ID、班级ID、用户角色和签名等。

- `SE_NODES`：Selenium Grid节点的URL列表，支持多个节点以实现分布式测试。

- `SE_NODE_SESSIONS`：每个节点上创建的浏览器会话数量。

- `BARRAGE_MESSAGES`：弹幕消息列表，包含20条预设消息，测试时会随机选择一条发送

```Python
import threading
import time
import random
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.remote.webdriver import WebDriver
from selenium.common.exceptions import NoSuchElementException, InvalidSelectorException


class LiveStreamTester:
    *"""直播平台压力测试工具"""*

*    *# 常量配置
    BASE_URL = "https://live.soa.com.cn:55555/#/login"
    URL_PARAMS = {
        "room_title": "Stress_Test",
        "room_id": "7626-5831-7513",
        "classId": "1939963504730525698",
        "user_role": "10",
        "sign": "U2FsdGVkX1+pLUjaZO8mdqoitBaOqupZuwf9veBmMbkV8OxkOapzZH9oaZH4GjjJ"
    }
    SE_NODES = [
        "http://172.22.1.93:4444",
        "http://172.22.1.93:4445"
    ]
    SE_NODE_SESSIONS = 30

    # 弹幕配置
    BARRAGE_MESSAGES = [
        "大家好！",
        "这个直播真精彩！",
        "主播讲得真好",
        "666666",
        "学到了学到了",
        "有人一起讨论吗？",
        "前排围观",
        "这个知识点很重要",
        "求解答",
        "感谢主播分享",
        "这个功能太实用了",
        "讲得太深入了，受益匪浅",
        "谁能帮忙解答下问题？",
        "没听明白，能再讲一遍吗？",
        "知识点都记下来了，感谢",
        "声音有点小，能调大点吗？",
        "弹幕刷起来！",
        "老师辛苦了，讲得真好",
        "这个案例很典型，学到了",
        "明天还会直播吗？",
    ]

    def __init__(self):
        *"""初始化测试器"""*
*        *self._setup_webdriver_options()
        self.drivers = []  # 存储所有已登录的driver实例
        self.usernames = []  # 存储所有用户名
        self.username_counter = 0  # 全局递增计数器
        self.instance_identifier = int(time.time() * 1000)  # 时间戳作为运行标识符,保证用户名唯一性

    def _setup_webdriver_options(self):
        *"""配置WebDriver选项"""*
*        *self.chrome_options = webdriver.ChromeOptions()
        self.chrome_options.add_experimental_option("detach", True)
        self.chrome_options.add_argument("ignore-certificate-errors")

    def _generate_username(self) -> str:
        *"""生成有序不重复的用户名"""*
*        *# 生成格式: 时间戳_自增编号
        username = f"虚拟用户_{self.instance_identifier}_{self.username_counter}"
        self.username_counter += 1
        return username

    def _construct_url(self) -> str:
        *"""参数化构建完整的测试URL"""*
*        *params = "&".join(f"{k}={v}" for k, v in self.URL_PARAMS.items())
        return f"{self.BASE_URL}?{params}"

    def _login(self, driver: WebDriver, username: str) -> bool:
        *"""执行登录流程"""*
*        *try:
            # 访问目标URL
            driver.get(self._construct_url())
            driver.implicitly_wait(10)
            time.sleep(1)  # 等待页面加载完成

            # 输入用户名
            username_input = self._find_element_safely(driver, "//div[4]/div/div/div/input")
            username_input.send_keys(username)

            # 点击登录按钮
            login_button = self._find_element_safely(driver, ".el-button")
            login_button.click()
            time.sleep(2)  # 等待登录完成

            print(f"用户 {username} 登录成功!")
            return True

        except Exception as e:
            print(f"用户 {username} 登录失败: {str(e)}")
            return False

    def _chat(self, driver: WebDriver, username: str, message: str = None):
        *"""在聊天框中输入指定消息并发送。如果未提供消息，则使用默认消息 """*
*        *try:
            if not message:
                message = f"用户 {username} 发来一条弹幕！"

            # 找到聊天输入框并输入消息
            chat_input = self._find_element_safely(driver, ".textarea-container:nth-child(2) > .textarea")
            chat_input.click()
            chat_input.send_keys(message)
            chat_input.send_keys(Keys.ENTER)

            print(f"用户 {username} 发送弹幕: '{message}'")

        except Exception as e:
            print(f"用户 {username} 聊天交互失败: {str(e)}")

    def _find_element_safely(self, driver: WebDriver, selector: str):
        *"""*
*        智能元素定位，支持XPath和CSS选择器*
*        """*
*        *try:
            if selector.startswith(("//", "./", "(")):
                return driver.find_element(By.XPATH, selector)
            return driver.find_element(By.CSS_SELECTOR, selector)
        except InvalidSelectorException as e:
            raise InvalidSelectorException(f"无效的选择器: '{selector}'. 错误详情: {str(e)}")
        except NoSuchElementException as e:
            raise NoSuchElementException(f"未找到元素: '{selector}'. 错误详情: {str(e)}")

    def login_all_users(self):
        *"""批量用户登陆"""*
*        *print(f"开始测试，直播间URL: {self._construct_url()}")
        total_users = len(self.SE_NODES) * self.SE_NODE_SESSIONS
        print(f"将创建 {total_users} 个浏览器用户实例")

        for node_index, node_url in enumerate(self.SE_NODES):
            for instance_num in range(self.SE_NODE_SESSIONS):
                username = self._generate_username()
                selenium_url = f"{node_url}/wd/hub"
                print(f"创建浏览器实例 (节点:{node_url})")

                # 通过Selenium Grid创建远程WebDriver实例
                try:
                    driver = webdriver.Remote(command_executor=selenium_url, options=self.chrome_options)
                    # driver = webdriver.Chrome(options=self.chrome_options) # 切换为本地测试
                    if self._login(driver, username):
                        self.drivers.append(driver)
                        self.usernames.append(username)
                except Exception as e:
                    print(f"创建WebDriver实例失败:{str(e)}")

    def run_barrage_test(self, test_type: str = "concurrent", count: int = 1):
        *"""*
*        执行弹幕测试的入口方法 ("concurrent":并发弹幕, "all":全员弹幕)*
*        """*
*        *if not self.drivers:
            print("没有已登录的用户，无法执行弹幕测试！")
            return
        print(f"\n开始弹幕压力测试，共有 {len(self.drivers)} 个已登录用户")

        if test_type == "concurrent":
            # 并发弹幕测试
            print(f"执行并发弹幕测试 ({count} 个用户同时发送)")
            self._send_concurrent_barrage(count)
        elif test_type == "all":
            # 全员弹幕测试
            print("执行全员弹幕测试 (所有用户依次发送，默认间隔1s)")
            self._send_all_barrage()
        else:
            print(f"未知测试类型: {test_type}")

    def _send_concurrent_barrage(self, user_count: int):
        *"""并发发送弹幕（使用多线程+时间戳同步实现毫秒级并发控制）"""*
*        *threads = []
        selected_drivers = random.sample(self.drivers, user_count)

        print(f"选中 {len(selected_drivers)} 个用户进行并发弹幕测试")

        # 设置同步开始时间
        start_time = time.time() + 3  # 3秒后开始

        for driver in selected_drivers:
            # 获取对应的用户名
            index = self.drivers.index(driver)
            username = self.usernames[index]

            # 随机选择一条弹幕消息
            message = random.choice(self.BARRAGE_MESSAGES)

            # 创建并启动线程
            t = threading.Thread(
                target=self._send_barrage_at_time,
                args=(driver, username, message, start_time)
            )
            threads.append(t)
            t.start()

        # 等待所有线程完成
        for t in threads:
            t.join()

        print(f"并发弹幕测试完成！{len(threads)}条弹幕已发送")

    def _send_barrage_at_time(self, driver: WebDriver, username: str, message: str, start_time: float):
        *"""指定时间内发送弹幕"""*
*        *# 等待直到开始时间
        while time.time() < start_time:
            time.sleep(0.01)
        # 发送弹幕
        self._chat(driver, username, message)

    def _send_all_barrage(self):
        *"""所有用户依次发送弹幕，统计成功率和发送数量"""*
*        *total_count = len(self.drivers)
        success_count = 0

        for i, driver in enumerate(self.drivers):
            username = self.usernames[i]
            message = random.choice(self.BARRAGE_MESSAGES)

            try:
                self._chat(driver, username, message)
                success_count += 1
                time.sleep(0.5)  # 短暂间隔
            except Exception as e:
                print(f"用户 {username} 弹幕发送失败: {str(e)}")

        success_rate = (success_count / total_count) * 100 if total_count > 0 else 0
        print(f"全员弹幕测试完成: 成功发送 {success_count}/{total_count} 条弹幕 (成功率: {success_rate:.2f}%)")

    def close_all_sessions(self):
        *"""关闭所有浏览器会话"""*
*        *print(f"\n关闭所有浏览器会话 ({len(self.drivers)} 个)")
        for driver in self.drivers:
            try:
                driver.quit()
            except Exception:
                pass
        self.drivers = []
        self.usernames = []
        print("所有会话已关闭")


if __name__ == "__main__":
    # 创建测试器实例
    tester = LiveStreamTester()
    tester.login_all_users()
    tester.run_barrage_test("concurrent", count=200)
    tester.run_barrage_test("all")
    tester.close_all_sessions()
```
